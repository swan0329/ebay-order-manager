it('keeps rounded corners, all edges, background toggle and shadow toggle',async()=>{
 const {createPreparedListingImage:render}=await import('../../outputs/card-layout-review-2026-09-11/engine/ebay-watermarked-images');
 const cfg={watermarkOpacity:0.04,watermarkLogoSize:50,watermarkGap:25,paddingTop:40,paddingBottom:40,paddingLeft:40,paddingRight:40};
 const source=await sharp({create:{width:300,height:500,channels:3,background:'red'}}).png().toBuffer();
 const base=await render(source,cfg,{});const pix=async(b:Buffer,x:number,y:number)=>sharp(b).extract({left:x,top:y,width:1,height:1}).removeAlpha().raw().toBuffer();
 // Contained card is 662px wide, centred at x=400; corners are transparent to white.
 expect((await pix(base,70,49))[1]).toBeGreaterThan(220);
 expect((await pix(base,400,50))[1]).toBeLessThan(20);
 const shadow=await render(source,{...cfg,shadowEnabled:true,shadowOpacity:0.8},{});
 expect(shadow.equals(base)).toBe(false);
 const bg=await sharp({create:{width:10,height:10,channels:3,background:'blue'}}).png().toBuffer();
 const off=await render(source,{...cfg,backgroundEnabled:false},{backgroundEligible:true,background:bg});
 const on=await render(source,{...cfg,backgroundEnabled:true},{backgroundEligible:true,background:bg});
 expect((await pix(off,0,0))[0]).toBeGreaterThan(220);expect((await pix(on,0,0))[0]).toBeLessThan(20);
 for(const angle of [0,90,33]){
  const marker=Buffer.from('<svg width="300" height="500"><rect width="300" height="500" fill="red"/><rect width="300" height="15" fill="lime"/><rect y="485" width="300" height="15" fill="blue"/><rect width="15" height="500" fill="yellow"/><rect x="285" width="15" height="500" fill="magenta"/></svg>');
  const out=await render(marker,{...cfg,imageRotation:angle,imageFlipHorizontal:true,imageZoom:100},{});
  const data=await sharp(out).removeAlpha().raw().toBuffer();
  for(const color of [[0,255,0],[0,0,255],[255,255,0],[255,0,255]]){let n=0;for(let i=0;i<data.length;i+=3)if(color.every((v,c)=>Math.abs(data[i+c]-v)<20))n++;expect(n).toBeGreaterThan(20);}
 }
});
